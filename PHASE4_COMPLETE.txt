   ```javascript
   window.require.config({
       paths: { 'vs': 'https://.../monaco-editor/.../vs' }

       const editor = monaco.editor.create(container, config);
   });
   ```

2. Excalidraw Scene Update
   ```javascript
   excalidrawRef.current.updateScene({
       elements: [...shapes, ...text, ...arrows],
       appState: { zoom, scrollX, scrollY, ... }
   });
   ```

   const { generateJsonFromResults, monacoEditorInstance } = useDiagram();
